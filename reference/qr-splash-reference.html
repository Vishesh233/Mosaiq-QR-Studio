    def mulberry32(seed):
        s = seed & 0xFFFFFFFF
        def rnd():
            nonlocal s
            s = (s + 0x6D2B79F5) & 0xFFFFFFFF
            t = ((s ^ (s >> 15)) * (1 | s)) & 0xFFFFFFFF
            t = ((t + (((t ^ (t >> 7)) * (61 | t)) & 0xFFFFFFFF)) & 0xFFFFFFFF) ^ t
            return ((t ^ (t >> 14)) & 0xFFFFFFFF) / 4294967296
        return rnd
